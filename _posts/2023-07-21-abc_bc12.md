​


### 아나콘다 터미널 내 환경 설치

```python
# 가상환경 내에서 권한을 체크
conda info —envs

# 파이썬 버전들을 확인
conda search python

# 선택한 파이썬 버전으로 가상환경 이름 생성
conda create —name py3_8_5 python=3.8.5

# 생성한 가상환경 권한 이동
conda activate py3_8_5

# 권한 이동 확인
conda info —envs

# 필요한 라이브러리 설치
pip install numpy
pip install pandas
pip install matplotlib
pip install beautifulsoup4
pip install selenium
pip install pymysql
pip install cryptography
pip install openpyxl
pip install xlrd
pip install ipykernel
pip install jupyter

python -m ipykernel install —user —name py3_8_5 —display "python3_8_5"

jupyter notebook
​
```
